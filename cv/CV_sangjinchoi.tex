\cvsection{RESEARCH EXPERIENCE}

\begin{itemize}[leftmargin=11pt,label=\textbullet,itemsep=4pt,parsep=2pt]
  \item \textbf{ML Systems}
    \begin{itemize}[leftmargin=0pt,label={},labelsep=0pt,itemsep=2pt,parsep=0pt]
      \item \textbf{CoSpec} (Sep 2023 \textendash\ Oct 2024)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item Efficient LLM serving system overcoming speculative decoding limitations in real-world scenarios, including high request rates and temperature sensitivity. It introduces Dynamic Colocation for smart GPU resource management across batch sizes and Selective Validation to minimize wasted computation and maintain accuracy.
        \end{itemize}

      \item \textbf{Hierarchical Drafting} (Aug 2024 \textendash\ Oct 2024)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item A novel lossless speculative decoding drafting approach that organizes various token sources into multiple databases in a hierarchical framework based on temporal locality. Ensures consistent, robust inference acceleration across diverse tasks and model sizes.
        \end{itemize}

      \item \textbf{EnvPipe} (May 2022 \textendash\ Jan 2023)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item Designed and implemented a framework for training large language models that significantly reduces energy consumption while maintaining minimal performance degradation.
          \item Optimized pipeline parallelism by strategically scheduling bubbles and selectively lowering SM frequency.
          \item Developed the system on top of DeepSpeed (\href{https://github.com/casys-kaist/EnvPipe}{https://github.com/casys-kaist/EnvPipe}).
        \end{itemize}
    \end{itemize}

  \item \textbf{Memory Systems}
    \begin{itemize}[leftmargin=0pt,label={},labelsep=0pt,itemsep=2pt,parsep=0pt]
      \item \textbf{HUVM} (Mar 2021 \textendash\ Jan 2022)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item Developed Hierarchical Unified Virtual Memory (HUVM), a system that creates the illusion of a unified virtual memory space for GPUs by utilizing temporarily idle memory of neighboring GPUs.
          \item Designed and implemented a novel memory manager, memHarvester, to efficiently harvest and utilize neighboring GPUs' memory via NVLink.
          \item Built a prototype system based on NVIDIA's driver (\href{https://github.com/casys-kaist/HUVM}{https://github.com/casys-kaist/HUVM}).
        \end{itemize}

      \item \textbf{TribuOS} (Sep 2019 \textendash\ Jun 2020)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item Contributed to the development of a disaggregated memory system designed for high availability.
          \item Implemented a memory server with support for 1 GB huge pages and an RDMA RPC recovery mechanism to enable faster recovery times.
        \end{itemize}
    \end{itemize}

  \item \textbf{AI with FPGA}
    \begin{itemize}[leftmargin=0pt,label={},labelsep=0pt,itemsep=2pt,parsep=0pt]
      \item \textbf{PRIMO} (Oct 2022 \textendash\ May 2023)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item Participated in developing a full-stack processing-in-DRAM emulation framework named PRIMO, the first emulation framework that can model and analyze DRAM-PIM for end-to-end ML inference.
          \item Collaborated with compiler and FPGA teams to implement a PIM driver that enables efficient execution of PIM operations by providing seamless integration between the software and hardware components.
        \end{itemize}

      \item \textbf{Accelerating DNN with FPGA} (Dec 2020 \textendash\ Dec 2021)
        \begin{itemize}[label=--,itemsep=0pt,parsep=0pt]
          \item Participated in developing an acceleration framework for DNNs using FPGA.
          \item Implemented functionality on top of the Xilinx driver to support encoding layers as FPGA instructions and quantizing models to an 8-bit fixed-point format to accelerate model inference.
        \end{itemize}
    \end{itemize}
\end{itemize}
